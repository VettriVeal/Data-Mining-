import os
import re
import boto3
from botocore.client import Config

DATA_DIR = r"C:\Users\ub02-glab-080\Downloads\data_2\data\sales"

ENDPOINT = "http://localhost:9002"
ACCESS_KEY = "minioadmin"
SECRET_KEY = "minioadmin123"
BUCKET = "annapurna"

s3 = boto3.client(
    "s3",
    endpoint_url=ENDPOINT,
    aws_access_key_id=ACCESS_KEY,
    aws_secret_access_key=SECRET_KEY,
    config=Config(signature_version="s3v4"),
    region_name="us-east-1"
)

pattern = re.compile(r"SALES_(S\d{2})_(\d{4})(\d{2})(\d{2})", re.IGNORECASE)

count = 0

for filename in os.listdir(DATA_DIR):

    if not filename.lower().endswith((".csv", ".parquet")):
        continue

    match = pattern.search(filename)

    if not match:
        print("SKIP:", filename)
        continue

    store_id, year, month, day = match.groups()

    object_key = (
        f"raw/sales/"
        f"year={year}/"
        f"month={month}/"
        f"store_id={store_id}/"
        f"{filename}"
    )

    local_path = os.path.join(DATA_DIR, filename)

    print(f"Uploading: {filename} -> {object_key}")

    s3.upload_file(
        local_path,
        BUCKET,
        object_key
    )

    count += 1

print()
print(f"Uploaded files: {count}")
print("Partitioning: year / month / store_id")
